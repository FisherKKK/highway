# Highway SIMD 快速参考手册

这是一份快速参考手册，涵盖 Highway 库最常用的 API 和模式。适合作为编程时的查阅资料。

## 📌 基本设置

### 包含头文件

```cpp
#include "hwy/highway.h"  // 主头文件，必须包含

namespace hwy {
namespace HWY_NAMESPACE {  // 必须在此命名空间内
// 你的代码
}  // namespace HWY_NAMESPACE
}  // namespace hwy
```

### 命名空间别名 (推荐)

```cpp
namespace hn = hwy::HWY_NAMESPACE;

// 使用别名简化代码
auto v = hn::Add(v1, v2);
```

## 🏷️ 类型系统

### 创建 Tag (描述符)

```cpp
// 全向量宽度（推荐，适应硬件）
const ScalableTag<float> df;        // 浮点向量
const ScalableTag<uint8_t> du8;     // 8位无符号整数向量
const ScalableTag<int32_t> di32;    // 32位有符号整数向量

// 固定最大通道数
const CappedTag<float, 4> d4f;      // 最多4个float
const CappedTag<uint8_t, 16> d16u8; // 最多16个uint8_t

// 精确通道数（N必须是2的幂）
const FixedTag<double, 2> d2d;      // 恰好2个double
const FixedTag<int16_t, 8> d8i16;   // 恰好8个int16_t
```

### 类型查询

```cpp
const ScalableTag<float> d;

// 获取通道数
size_t N = Lanes(d);                // 向量有多少个元素
size_t maxN = MaxLanes(d);          // 编译时最大通道数

// 获取类型信息
using T = TFromD<decltype(d)>;      // 通道类型 (float)
using V = Vec<decltype(d)>;         // 向量类型
using M = Mask<decltype(d)>;        // 掩码类型
```

### 类型转换 Tag

```cpp
const ScalableTag<uint8_t> du8;

// Rebind: 改变通道类型，保持通道数
const Rebind<uint32_t, decltype(du8)> du32;  // uint8_t -> uint32_t, 通道数不变

// Repartition: 改变通道类型，保持总字节数
const Repartition<uint16_t, decltype(du8)> du16r;  // uint8_t -> uint16_t, 字节数不变

// Half: 通道数减半
const Half<decltype(du8)> du8h;

// Twice: 通道数加倍
const Twice<decltype(du8)> du8t;
```

## 📦 加载与存储

### 基本加载/存储

```cpp
const ScalableTag<float> d;
float* data = /* ... */;

// 对齐加载 (地址必须对齐到向量大小)
auto v = Load(d, data);

// 非对齐加载 (任意地址，但稍慢)
auto v = LoadU(d, data);

// 对齐存储
Store(v, d, data);

// 非对齐存储
StoreU(v, d, data);
```

### 特殊加载/存储

```cpp
// 加载单个值并广播到所有通道
auto v = Set(d, 3.14f);              // [3.14, 3.14, 3.14, ...]

// 零向量
auto v = Zero(d);                    // [0, 0, 0, ...]

// 未定义值（用于临时变量）
auto v = Undefined(d);

// 连续整数 [0, 1, 2, 3, ...]
auto v = Iota(d, 0);

// 条件存储（仅存储掩码为真的通道）
BlendedStore(v, mask, d, data);

// 掩码加载（掩码为假的通道填充为0）
auto v = MaskedLoad(mask, d, data);
```

### Gather/Scatter (索引访问)

```cpp
const ScalableTag<float> d;
const ScalableTag<int32_t> di;

float data[1000];
int32_t indices[MaxLanes(d)] = {0, 5, 10, 15, ...};

// Gather: 根据索引加载
auto idx = Load(di, indices);
auto v = GatherIndex(d, data, idx);

// Scatter: 根据索引存储
ScatterIndex(v, d, data, idx);
```

## 🧮 算术运算

### 基本算术

```cpp
auto sum = Add(a, b);                // a + b
auto diff = Sub(a, b);               // a - b
auto prod = Mul(a, b);               // a * b
auto quot = Div(a, b);               // a / b (仅浮点)
auto neg = Neg(a);                   // -a
auto abs_val = Abs(a);               // |a|
```

### 饱和算术 (整数)

```cpp
auto sum = SaturatedAdd(a, b);       // a + b (饱和到类型范围)
auto diff = SaturatedSub(a, b);      // a - b (饱和到0或最大值)
```

### 融合乘加 (FMA)

```cpp
auto result = MulAdd(a, b, c);       // a * b + c
auto result = MulSub(a, b, c);       // a * b - c
auto result = NegMulAdd(a, b, c);    // -(a * b) + c = c - a*b
auto result = NegMulSub(a, b, c);    // -(a * b) - c
```

### 最小/最大值

```cpp
auto minimum = Min(a, b);            // min(a, b)
auto maximum = Max(a, b);            // max(a, b)
auto clamped = Clamp(v, lo, hi);     // clamp(v, lo, hi)
```

### 舍入 (浮点)

```cpp
auto rounded = Round(v);             // 四舍五入到最近整数
auto ceiled = Ceil(v);               // 向上取整
auto floored = Floor(v);             // 向下取整
auto truncated = Trunc(v);           // 向零截断
```

## 🔀 逻辑运算

### 位运算

```cpp
auto and_result = And(a, b);         // a & b
auto or_result = Or(a, b);           // a | b
auto xor_result = Xor(a, b);         // a ^ b
auto not_result = Not(a);            // ~a
auto andnot = AndNot(a, b);          // (~a) & b
```

### 位移

```cpp
// 立即数位移（编译时常量）
auto left = ShiftLeft<3>(v);         // v << 3
auto right = ShiftRight<3>(v);       // v >> 3 (逻辑右移)
auto arith_right = ShiftRightSame<3>(v);  // v >> 3 (算术右移)

// 向量位移（每个通道可以不同）
auto left = Shl(v, shift_amounts);
auto right = Shr(v, shift_amounts);
```

### 旋转

```cpp
auto rotated = RotateRight<3>(v);    // 循环右移3位
```

```cpp
auto eq_mask = Eq(a, b);             // a == b

```

### 测试特殊值 (浮点)

```cpp
auto is_nan = IsNaN(v);              // 是否为 NaN
```

## 🎭 掩码操作

### 掩码运算

```cpp
auto mask_and = And(m1, m2);         // m1 & m2
auto mask_or = Or(m1, m2);           // m1 | m2
auto mask_xor = Xor(m1, m2);         // m1 ^ m2
auto mask_not = Not(m1);             // ~m1
```

### 掩码测试

```cpp
bool all_true = AllTrue(d, mask);    // 所有通道都为真?
bool all_false = AllFalse(d, mask);  // 所有通道都为假?
size_t count = CountTrue(d, mask);   // 有多少个通道为真?
```

### 条件选择

```cpp
// 掩码为真选 a，为假选 b
auto result = IfThenElse(mask, a, b);

// 掩码为真选 a，为假选 0
auto result = IfThenElseZero(mask, a);

// 掩码为真选 0，为假选 b
auto result = IfThenZeroElse(mask, b);
```

### 掩码与向量转换

```cpp
// 掩码转向量 (真->全1, 假->全0)
auto v = VecFromMask(d, mask);

// 向量转掩码 (根据符号位)
auto mask = MaskFromVec(v);
```

### 创建掩码

```cpp
// 前N个通道为真
auto mask = FirstN(d, count);

// 特定位置的掩码
uint64_t bits = 0b1010;  // 第1和第3个通道为真
auto mask = LoadMaskBits(d, bits);

// 存储掩码为位
StoreMaskBits(d, mask, &bits);
```

## 🔄 通道排列

### 广播

```cpp
// 广播单个值到所有通道
auto v = Set(d, value);

// 广播向量的第i个通道
auto v = Broadcast<2>(v_in);         // 所有通道 = v_in[2]
```

### 交错 (Interleave)

```cpp
auto lo = InterleaveLower(a, b);     // [a0,b0,a1,b1, ...]
auto hi = InterleaveUpper(a, b);     // [a4,b4,a5,b5, ...] (假设8通道)
```

### 拼接与组合

```cpp
// 组合两个半向量
auto full = Combine(d, hi_half, lo_half);

// 拼接
auto result = ConcatLowerLower(d, a, b);    // [b_lo, a_lo]
auto result = ConcatUpperUpper(d, a, b);    // [b_hi, a_hi]
auto result = ConcatLowerUpper(d, a, b);    // [b_hi, a_lo]
auto result = ConcatUpperLower(d, a, b);    // [b_lo, a_hi]
```

### 逆序

```cpp
auto rev = Reverse(d, v);            // 完全逆序 [3,2,1,0]
auto rev2 = Reverse2(d, v);          // 两两逆序 [1,0,3,2]
auto rev4 = Reverse4(d, v);          // 四个一组逆序
```

### 查表 (Table Lookup)

```cpp
// 通道级查表
auto result = TableLookupLanes(v, indices);

// 字节级查表
auto result = TableLookupBytes(v, byte_indices);
```

### 滑动

```cpp
auto slid = SlideUpLanes(d, v, 2);   // 向上滑动2个通道
auto slid = SlideDownLanes(d, v, 2); // 向下滑动2个通道
```

### Compress/Expand

```cpp
// 压缩：保留掩码为真的元素
auto compressed = Compress(v, mask);

// 压缩并存储
size_t count = CompressStore(v, mask, d, dest);

// 展开（Compress 的逆操作）
auto expanded = Expand(v, mask);
```

## 🔢 类型转换

### 提升/降级

```cpp
const ScalableTag<uint8_t> du8;
const ScalableTag<uint16_t> du16;

// 提升 (扩展位宽)
auto wider = PromoteTo(du16, v_u8);  // uint8_t -> uint16_t

// 降级 (缩减位宽，可能截断)
auto narrower = DemoteTo(du8, v_u16); // uint16_t -> uint8_t
```

### 类型转换

```cpp
const ScalableTag<float> df;
const ScalableTag<int32_t> di32;

// 整数 <-> 浮点
auto f = ConvertTo(df, v_i32);       // int32_t -> float
auto i = ConvertTo(di32, v_f);       // float -> int32_t (截断)

// 有符号 <-> 无符号 (仅重新解释)
auto u = BitCast(du32, v_i32);       // int32_t -> uint32_t
```

### 截断

```cpp
// 截断到半宽度
auto truncated = TruncateTo(du8, v_u32);  // uint32_t -> uint8_t
```

## 📉 归约操作

### 水平归约

```cpp
auto sum = ReduceSum(d, v);          // v[0] + v[1] + ... + v[N-1]
auto min_val = ReduceMin(d, v);      // min(v[0], v[1], ..., v[N-1])
auto max_val = ReduceMax(d, v);      // max(v[0], v[1], ..., v[N-1])
```

### 获取单个通道

```cpp
// 仅适用于单通道或 Scalar 目标
T value = GetLane(v);                // 获取第0个通道的值
```

### 全向量归约

```cpp
// 所有通道求和（返回向量，每个通道都是总和）
auto sum_vec = SumOfLanes(d, v);

// 所有通道最小值
auto min_vec = MinOfLanes(d, v);

// 所有通道最大值
auto max_vec = MaxOfLanes(d, v);
```

## 🔁 循环模式

### 基本循环（对齐数据）

```cpp
const ScalableTag<float> d;
const size_t N = Lanes(d);

for (size_t i = 0; i < count; i += N) {
  auto v = Load(d, &data[i]);
  // 处理 v
  Store(result, d, &output[i]);
}
```

### 处理余项（方法1：重叠）

```cpp
for (size_t i = 0; i < count; i += N) {
  size_t idx = HWY_MIN(i, count >= N ? count - N : 0);
  auto v = Load(d, &data[idx]);
  // 处理 v
  Store(result, d, &output[idx]);
}
```

### 处理余项（方法2：掩码）

```cpp
size_t i = 0;
for (; i + N <= count; i += N) {
  auto v = Load(d, &data[i]);
  Store(result, d, &output[i]);
}

// 处理剩余元素
if (i < count) {
  auto mask = FirstN(d, count - i);
  auto v = MaskedLoad(mask, d, &data[i]);
  BlendedStore(result, mask, d, &output[i]);
}
```

### 使用 Transform (推荐)

```cpp
#include "hwy/contrib/algo/transform-inl.h"

// 一元变换
Transform(d, input, count, output, [](auto d, auto v) {
  return Mul(v, v);  // 平方
});

// 二元变换
Transform2(d, input1, input2, count, output, [](auto d, auto v1, auto v2) {
  return Add(v1, v2);  // 相加
});
```

## 💡 性能优化提示

1. **优先使用 `ScalableTag`**: 充分利用硬件能力
2. **对齐内存**: 使用 `AlignedAlloc` 或确保数据对齐
3. **减少分支**: 使用掩码和 `IfThenElse` 代替 if-else
4. **批量处理**: 一次处理多个向量以隐藏延迟
5. **使用 FMA**: `MulAdd` 比单独的 `Mul` + `Add` 快
6. **避免归约**: 水平操作（如 `ReduceSum`）较慢
7. **预取数据**: 在大数据集上使用 `Prefetch`
