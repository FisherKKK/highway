## 8.5 比较操作实现

### 比较

```cpp
// EQ: 等于
template <typename T, size_t N>
Mask128<T, N> EQ(const Vec128<T, N> a, const Vec128<T, N> b) {
#if HWY_TARGET == HWY_SSE2
  return Mask128<T, N>{_mm_cmpeq_epi8(a.raw, b.raw)};  // 8-bit
  return Mask128<T, N>{_mm_cmpeq_epi16(a.raw, b.raw)};  // 16-bit
  return Mask128<T, N>{_mm_cmpeq_epi32(a.raw, b.raw)};  // 32-bit
  return Mask128<T, N>{_mm_cmpeq_epi64(a.raw, b.raw)};  // 64-bit

  // 浮点
  return Mask128<T, N>{_mm_cmpeq_ps(a.raw, b.raw)};    // float
  return Mask128<T, N>{_mm_cmpeq_pd(a.raw, b.raw)};    // double
#endif
}

// LT: 小于
template <typename T, size_t N>
Mask128<T, N> LT(const Vec128<T, N> a, const Vec128<T, N> b) {
#if HWY_TARGET == HWY_SSE2
  return Mask128<T, N>{_mm_cmplt_epi8(a.raw, b.raw)};  // 有符号 8-bit
  return Mask128<T, N>{_mm_cmplt_epi16(a.raw, b.raw)}; // 有符号 16-bit
  return Mask128<T, N>{_mm_cmplt_epi32(a.raw, b.raw)}; // 有符号 32-bit
  return Mask128<T, N>{_mm_cmplt_ps(a.raw, b.raw)};     // 浮点
#endif
}
```

### 掩码操作

```cpp
// 检查掩码是否全为真
template <typename T, size_t N>
bool AllTrue(Simd<T, N, 0> d, const Mask128<T, N> mask) {
#if HWY_TARGET == HWY_SSE2
  // _mm_movemask_epi8 提取掩码位
  int32_t mask_bits = _mm_movemask_epi8(mask.raw);

  // 检查是否所有位都设置了
  constexpr size_t kNumBits = N * sizeof(T);
  return mask_bits == (1 << kNumBits) - 1;
#endif
}

// 计算掩码中真值的数量
template <typename T, size_t N>
size_t CountTrue(Simd<T, N, 0> d, const Mask128<T, N> mask) {
#if HWY_TARGET == HWY_SSE2
  int32_t mask_bits = _mm_movemask_epi8(mask.raw);
  return PopCount(static_cast<uint32_t>(mask_bits));
#endif
}
```

## 8.8 规约操作实现

### SumOfLanes

```cpp
// 求和所有通道
template <typename T, size_t N>
Vec128<T, N> SumOfLanes(Simd<T, N, 0> d, const Vec128<T, N> v) {
#if HWY_TARGET == HWY_SSE2
  // 水平相加：[a,b,c,d] -> [a+b, c+d]
  auto hadd = _mm_hadd_epi16(v.raw, v.raw);
  // 再次水平相加：[a+b+c+d, ...]
  hadd = _mm_hadd_epi16(hadd, hadd);
  // ... 根据通道数重复
  return Vec128<T, N>{hadd};
#endif
}

// 128-bit float 版本
template <size_t N>
Vec128<float, N> SumOfLanes(Simd<float, N, 0> d, const Vec128<float, N> v) {
#if HWY_TARGET == HWY_SSE2
  // [a,b,c,d] -> [a+b, c+d]
  auto hadd = _mm_hadd_ps(v.raw, v.raw);
  // [a+b+c+d, a+b+c+d]
  hadd = _mm_hadd_ps(hadd, hadd);
  // [sum, sum, ...]（shuffle）
  auto sum = _mm_shuffle_ps(hadd, hadd, 0);  // 提取第一个元素
  auto result = _mm_shuffle_ps(sum, sum, 0);  // 广播到所有通道
  return Vec128<float, N>{result};
#endif
}
```

## 8.9 AVX2 特定操作

### 256-bit 特性

```cpp
// x86_256-inl.h

// Combine：两个 128-bit -> 一个 256-bit
template <typename T, size_t N>
Vec256<T, N> Combine(Simd<T, N, 0> d,
                     const Vec128<T, N/2> lo,
                     const Vec128<T, N/2> hi) {
#if HWY_TARGET == HWY_AVX2
  // _mm256_set_m128i: 设置低 128-bit
  // _mm256_inserti128_si256: 插入高 128-bit
  auto result = _mm256_set_m128i(lo.raw);
  result = _mm256_inserti128_si256(result, hi.raw, 1);
  return Vec256<T, N>{result};
#endif
}

// LowerHalf / UpperHalf
template <typename T, size_t N>
Vec128<T, N/2> LowerHalf(Simd<T, N/2, -1> d,
                          const Vec256<T, N> v) {
#if HWY_TARGET == HWY_AVX2
  // _mm256_castsi256_si128: 提取低 128-bit
  return Vec128<T, N/2>{_mm256_castsi256_si128(v.raw)};
#endif
}

template <typename T, size_t N>
Vec128<T, N/2> UpperHalf(Simd<T, N/2, -1> d,
                          const Vec256<T, N> v) {
#if HWY_TARGET == HWY_AVX2
  // _mm256_extracti128_si256: 提取高 128-bit
  return Vec128<T, N/2>{_mm256_extracti128_si256(v.raw, 1)};
#endif
}
```
