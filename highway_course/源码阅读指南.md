```cpp
HWY_API size_t PopCount(uint64_t x) {
#if HWY_COMPILER_GCC || HWY_COMPILER_CLANG
  return static_cast<size_t>(__builtin_popcountll(x));
#elif HWY_COMPILER_MSVC && defined(_M_X64)
  return _mm_popcnt_u64(x);
#else
  // 软件回退实现
  x = x - ((x >> 1) & 0x5555555555555555ULL);
  x = (x & 0x3333333333333333ULL) + ((x >> 2) & 0x3333333333333333ULL);
  x = (x + (x >> 4)) & 0x0F0F0F0F0F0F0F0FULL;
  return (x * 0x0101010101010101ULL) >> 56;
#endif
}
```

## 3. 关键实现剖析

### 3.1 Load/Store 的实现

以 x86 SSE2 为例，看看 Load 是如何实现的。

**文件**: `hwy/ops/x86_128-inl.h`

```cpp
// 行 500-600: Load 的多个重载

// 1. 对齐加载 - 整数
template <size_t N>
HWY_API Vec128<uint8_t, N> Load(Simd<uint8_t, N, 0> /* tag */,
                                 const uint8_t* HWY_RESTRICT aligned) {
  return Vec128<uint8_t, N>(_mm_load_si128(
      reinterpret_cast<const __m128i*>(aligned)));
}

// 2. 非对齐加载 - 整数
template <size_t N>
HWY_API Vec128<uint8_t, N> LoadU(Simd<uint8_t, N, 0> /* tag */,
                                  const uint8_t* HWY_RESTRICT p) {
  return Vec128<uint8_t, N>(_mm_loadu_si128(
      reinterpret_cast<const __m128i*>(p)));
}

// 3. 对齐加载 - 浮点
template <size_t N>
HWY_API Vec128<float, N> Load(Simd<float, N, 0> /* tag */,
                               const float* HWY_RESTRICT aligned) {
  return Vec128<float, N>(_mm_load_ps(aligned));
}

// 4. 部分加载（小于完整向量）
template <size_t N, HWY_IF_LE64(uint8_t, N)>
HWY_API Vec128<uint8_t, N> Load(Simd<uint8_t, N, 0> d,
                                 const uint8_t* HWY_RESTRICT p) {
  // 对于小向量，使用特殊处理
  if (N == 1) {
    return Vec128<uint8_t, N>(_mm_cvtsi32_si128(*p));
  }
  // ... 其他大小的处理
}
```

**关键观察**:
1. **函数重载**: 根据类型和大小选择不同实现
2. **类型转换**: 指针需要转换为 intrinsic 期望的类型
3. **小向量处理**: N < 16 时需要特殊处理

### 3.3 Compress 的实现（复杂操作）

Compress 是一个复杂操作：根据掩码压缩向量。

**示例**: `[1, 2, 3, 4]` + `mask = [T, F, T, F]` → `[1, 3, ?, ?]`

#### AVX-512 版本（硬件支持）

```cpp
// 文件: hwy/ops/x86_512-inl.h, 行 2000-2100

HWY_API Vec512<float> Compress(Vec512<float> v, Mask512<float> m) {
  // AVX-512 有专用指令 VCOMPRESS
  return Vec512<float>(_mm512_mask_compress_ps(
      _mm512_setzero_ps(),  // 未使用的通道填0
      m.raw,                 // 掩码
      v.raw                  // 输入向量
  ));
}
```

#### AVX2 版本（查表法）

```cpp
// 文件: hwy/ops/x86_256-inl.h, 行 3000-3200

HWY_API Vec256<float> Compress(Vec256<float> v, Mask256<float> m) {
  // AVX2 没有硬件 Compress，使用查表法
  alignas(32) static constexpr uint32_t table[256][8] = {
    // 预计算的索引表
    // table[mask_bits] = 重排索引
    {0, 1, 2, 3, 4, 5, 6, 7},  // mask = 11111111
    {1, 2, 3, 4, 5, 6, 7, 0},  // mask = 11111110
    // ... 256 个条目
  };

  const uint32_t mask_bits = MoveMask(m);  // 掩码转整数
  const auto indices = Load(Full256<uint32_t>(), &table[mask_bits][0]);

  return TableLookupLanes(v, indices);  // 使用查表重排
}
```

#### Scalar 版本（标量循环）

```cpp
// 文件: hwy/ops/scalar-inl.h, 行 1500-1550

template <typename T>
HWY_API Vec1<T> Compress(Vec1<T> v, Mask1<T> m) {
  // 标量版本很简单：掩码为真就返回值，否则返回未定义
  return m.raw ? v : Vec1<T>(T{});
}
```

**关键观察**:
1. **分层实现**: 硬件支持 > 查表 > 标量
2. **性能差异**: AVX-512 最快，AVX2 中等，Scalar 最慢
3. **正确性**: 三种实现语义相同

ARM NEON 使用宏来减少代码重复。

```cpp
// 行 100-200: 宏定义

// 1. 定义操作的元数据
#define HWY_NEON_BUILD_TPL_HWY_ADD
#define HWY_NEON_BUILD_RET_HWY_ADD(type, size) Vec128<type, size>
#define HWY_NEON_BUILD_PARAM_HWY_ADD(type, size) \

HWY_NEON_DEF_FUNCTION_ALL_TYPES(Add, vadd)
//                              ^    ^
//                              |    NEON intrinsic 前缀
//                              Highway 操作名

// 3. HWY_NEON_DEF_FUNCTION_ALL_TYPES 的展开
// 这个宏会生成：
//
// Vec128<int8_t> Add(Vec128<int8_t> a, Vec128<int8_t> b) {
//   return Vec128<int8_t>(vaddq_s8(a.raw, b.raw));
// }
//
// Vec128<int16_t> Add(Vec128<int16_t> a, Vec128<int16_t> b) {
//   return Vec128<int16_t>(vaddq_s16(a.raw, b.raw));
// }
//
// Vec128<int32_t> Add(Vec128<int32_t> a, Vec128<int32_t> b) {
//   return Vec128<int32_t>(vaddq_s32(a.raw, b.raw));
// }
//
// ... 等等，为所有类型生成
```

**NEON intrinsics 命名规则**:
```
vadd  q  _s  8
 ^    ^   ^  ^
 |    |   |  位宽 (8, 16, 32, 64)
 |    |   类型 (s=有符号, u=无符号, f=浮点)
 |    向量大小 (q=128位, 无q=64位)
 操作 (add, mul, sub, etc.)
```

**阅读建议**:
1. 先理解宏系统的原理
2. 手动展开几个宏，看生成的代码
3. 对比 x86 的手写版本，理解两种方法的权衡

### 4.2 ARM SVE 的可伸缩向量

SVE 的向量长度在运行时确定。

**文件**: `hwy/ops/arm_sve-inl.h`

```cpp
// 行 200-300: SVE 向量定义

template <typename T>
struct VecSVE {
  using Raw = /* SVE intrinsic 类型，如 svfloat32_t */;
  Raw raw;

  // 关键：不存储通道数！
  // 通道数在运行时由 svcntX() 查询
};

// 行 500-600: Load 实现
template <typename T>
HWY_API VecSVE<T> Load(ScalableTag<T> /* tag */, const T* p) {
  // SVE Load: 不需要指定通道数
  return VecSVE<T>(svld1(svptrue<T>(), p));
  //                      ^
  //                      全真谓词（加载所有通道）
}

// 行 1000-1100: 查询通道数
template <typename T>
HWY_API size_t Lanes(ScalableTag<T> /* tag */) {
  // 运行时查询
  return svcntb() / sizeof(T);
  //     ^
  //     返回向量字节数
}
```

**关键概念**:
1. **谓词 (Predicate)**: SVE 的掩码，用于控制哪些通道活跃
2. **可伸缩**: 同一份代码可以在不同向量长度的 CPU 上运行
3. **性能**: 理论上可以利用未来更宽的向量硬件

### 5.1 VQSort - 生产级排序算法

**文件**: `hwy/contrib/sort/vqsort-inl.h` (86KB)

#### 算法概述

VQSort 使用三种技术：
1. **Sorting Networks**: 固定比较序列（2-16 元素）
2. **Vectorized Partition**: SIMD 加速的分区
3. **Quicksort**: 递归排序

#### 关键代码：Sorting Network

```cpp
// 文件: hwy/contrib/sort/sorting_networks-inl.h, 行 100-200

// 2个元素的排序网络
template <class D, class V>
HWY_INLINE void Sort2(D d, V& a, V& b) {
  const V min_ab = Min(a, b);
  const V max_ab = Max(a, b);
  a = min_ab;
  b = max_ab;
}

// 4个元素的排序网络（Batcher's Odd-Even Merge）
template <class D, class V>
HWY_INLINE void Sort4(D d, V& a, V& b, V& c, V& d) {
  // 第1层：两两比较
  Sort2(d, a, c);
  Sort2(d, b, d);

  // 第2层：交叉比较
  Sort2(d, a, b);
  Sort2(d, c, d);

  // 第3层：中间比较
  Sort2(d, b, c);
}

// 8个元素、16个元素的网络类似，但更复杂
```

**为什么快？**:
1. **无分支**: 比较序列固定，CPU 可以完美预测
2. **向量化**: 多个元素并行比较
3. **缓存友好**: 数据局部性好

#### 关键代码：向量化分区

```cpp
// 文件: vqsort-inl.h, 行 2000-2200

template <class D, class V>
HWY_NOINLINE void Partition(D d, V* HWY_RESTRICT keys, size_t count,
                             V pivot, size_t& left_size) {
  const size_t N = Lanes(d);

  V* HWY_RESTRICT left = keys;
  V* HWY_RESTRICT right = keys + count - N;

  while (left + N <= right) {
    // 加载两端的数据
    auto v_left = Load(d, left);
    auto v_right = Load(d, right);

    // 创建掩码
    auto mask_left = Lt(v_left, pivot);   // 左边小于pivot的
    auto mask_right = Ge(v_right, pivot);  // 右边大于等于pivot的

    // Compress: 压缩需要移动的元素
    auto to_right = Compress(v_left, Not(mask_left));
    auto to_left = Compress(v_right, Not(mask_right));

    // 存储
    BlendedStore(to_left, mask_right, d, left);
    BlendedStore(to_right, mask_left, d, right);

    // 更新指针
    left += CountTrue(d, mask_left);
    right -= CountTrue(d, mask_right);
  }

  // 处理余项...

  left_size = left - keys;
}
```

```cpp
// 行 1000-1200: Sin 使用多项式近似

template <class D, class V>
HWY_NOINLINE V CallSin(D d, V x) {
  // 1. 范围规约：x -> [-π/2, π/2]
  //    使用 x = x - 2πk，k 是最接近的整数

  // 计算 k = round(x / (2π))
  const V kInv2Pi = Set(d, 0.159154943091895f);  // 1/(2π)
  const V k = Round(Mul(x, kInv2Pi));

  // x = x - k * 2π（精确计算，使用 FMA）
  const V k2Pi_hi = Set(d, -6.28125f);       // 高精度部分
  const V k2Pi_lo = Set(d, -0.00195312f);    // 低精度部分
  x = MulAdd(k, k2Pi_hi, x);
  x = MulAdd(k, k2Pi_lo, x);

  // 2. 多项式近似（minimax 多项式）
  // sin(x) ≈ x + c3*x^3 + c5*x^5 + c7*x^7 + c9*x^9
  const V c9 = Set(d, 2.60194326e-6f);
  const V c7 = Set(d, -1.98125763e-4f);
  const V c5 = Set(d, 8.33255797e-3f);
  const V c3 = Set(d, -1.66665134e-1f);

  const V x2 = Mul(x, x);  // x^2
  const V x3 = Mul(x2, x); // x^3

  // Horner 法则求值
  auto poly = c9;
  poly = MulAdd(poly, x2, c7);  // c9*x^2 + c7
  poly = MulAdd(poly, x2, c5);  // (c9*x^2 + c7)*x^2 + c5
  poly = MulAdd(poly, x2, c3);  // ...
  poly = Mul(poly, x3);         // poly * x^3

  // 最终结果
  return MulAdd(poly, x, x);    // poly*x^3*x + x
}
```

**关键技术**:
1. **范围规约**: 将输入映射到 [-π/2, π/2]
2. **多项式近似**: 使用泰勒级数或 minimax 多项式
3. **Horner 法则**: 减少乘法次数
4. **FMA**: 提高精度和性能

**精度**:
- 相对误差 < 1e-6（float）
- 足够大多数应用使用

**性能**:
- 比标准库 `std::sin` 快 5-10x
- 略微降低精度（可接受）

```cpp
// 实用的调试函数
template <class D>
void PrintVec(D d, Vec<D> v, const char* name) {
  using T = TFromD<D>;
  const size_t N = Lanes(d);

  HWY_ALIGN T lanes[MaxLanes(d)];
  Store(v, d, lanes);

  std::fprintf(stderr, "%s = [", name);
  for (size_t i = 0; i < N; ++i) {
    if constexpr (IsSame<T, float>() || IsSame<T, double>()) {
      std::fprintf(stderr, "%.6f", static_cast<double>(lanes[i]));
    } else {
      std::fprintf(stderr, "%lld", static_cast<long long>(lanes[i]));
    }

    if (i + 1 < N) std::fprintf(stderr, ", ");
  }
  std::fprintf(stderr, "]\n");
}

// 使用
auto v = Load(d, data);
PrintVec(d, v, "v after load");
v = Add(v, v);
PrintVec(d, v, "v after add");
```
